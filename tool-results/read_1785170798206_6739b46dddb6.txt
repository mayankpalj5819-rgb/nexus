   155→interface FetchPostsOptions {
   156→  sort?: FeedSort;
   157→  topicId?: string;
   158→  authorId?: string;
   159→  bookmarkedBy?: string;
   160→  limit?: number;
   161→  offset?: number;

   165→export async function fetchPosts(opts: FetchPostsOptions = {}): Promise<Post[]> {
   166→  if (!supabase) return [];
   167→  const { sort = "latest", topicId, authorId, bookmarkedBy, limit = 20, offset = 0, currentUserId } = opts;
   168→
   169→  let query = supabase.from("posts").select(`
   170→      id, author_id, title, preview, content, images, tags, views,
   171→      removed, removed_reason, created_at, updated_at,
   172→      upvote_count:post_votes!post_votes_post_id_fkey(count),
   173→      downvote_count:post_votes!post_votes_post_id_fkey(count),
   174→      comment_count:comments!comments_post_id_fkey(count),
   175→      post_topics(topic_id)
   176→    `);
   177→
   178→  // Joins via post_topics
   179→  if (topicId) {
   180→    // Use a sub-filter — Supabase JS supports filtering on related tables
   181→    query = supabase
   182→      .from("posts")
   183→      .select(`
   184→        id, author_id, title, preview, content, images, tags, views,
   185→        removed, removed_reason, created_at, updated_at,
   186→        post_topics!inner(topic_id)
   187→      `)
   188→      .eq("post_topics.topic_id", topicId);
   189→  }
   190→
   191→  if (authorId) query = query.eq("author_id", authorId);
   192→
   193→  // Note: bookmarkedBy filter is handled separately below because we can't
   194→  // directly filter by a foreign key on the bookmarks table from posts.
   195→
   196→  let postsQuery = query.eq("removed", false);
   197→
   198→  // For "week" filter, restrict to last 7 days
   199→  if (sort === "week") {
   200→    const weekAgo = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000).toISOString();
   201→    postsQuery = postsQuery.gte("created_at", weekAgo);
   202→  }
   203→
   204→  if (sort === "latest") {
   205→    postsQuery = postsQuery.order("created_at", { ascending: false });
   206→  } else if (sort === "popular") {
   207→    // Popular = most upvotes — we'll sort client-side after fetching counts
   208→    postsQuery = postsQuery.order("created_at", { ascending: false }).limit(limit * 3);
   209→  } else {
   210→    // trending, week — fetch recent, sort client-side
   211→    postsQuery = postsQuery.order("created_at", { ascending: false }).limit(limit * 3);
   212→  }
   213→
   214→  postsQuery = postsQuery.range(offset, offset + limit - 1);
   215→
   216→  const { data, error } = await postsQuery;
   217→  if (error) {
   218→    console.error("fetchPosts error:", error);
   219→    return [];
   220→  }
   221→
   222→  // If filtering by bookmarked posts, fetch bookmarked post IDs first
   223→  if (bookmarkedBy) {
   224→    const { data: bookmarkData } = await supabase
   225→      .from("bookmarks")
   226→      .select("post_id")
   227→      .eq("user_id", bookmarkBy);
   228→    const bookmarkedIds = (bookmarkData ?? []).map((b: { post_id: string }) => b.post_id);
   229→    const filtered = (data ?? []).filter((p: { id: string }) => bookmarkedIds.includes(p.id));
   230→    return transformPosts(filtered, sort, limit, currentUserId);
   231→  }
   232→
   233→  return transformPosts(data ?? [], sort, limit, currentUserId);
   234→}

   236→async function transformPosts(
   237→  rows: Record<string, unknown>[],
   238→  sort: FeedSort,
   239→  limit: number,
   240→  currentUserId?: string
   241→): Promise<Post[]> {
   242→  if (rows.length === 0) return [];
   243→
   244→  const authorIds = [...new Set(rows.map((r) => r.author_id as string))];
   245→  const postIds = rows.map((r) => r.id as string);
   246→
   247→  // Fetch authors
   248→  const { data: authors } = await supabase!
   249→    .from("users")
   250→    .select("*")
   251→    .in("id", authorIds);
   252→  const authorMap = new Map<string, Profile>(
   253→    (authors ?? []).map((a: Profile) => [a.id, a])
   254→  );
   255→
   256→  // Fetch topics for all posts
   257→  const { data: postTopics } = await supabase!
   258→    .from("post_topics")
   259→    .select("post_id, topic_id, topics!inner(id, name, slug, description, banner, color, icon, parent_id, post_count, created_at)")
   260→    .in("post_id", postIds);
   261→  const topicMap = new Map<string, Topic[]>();
   262→  (postTopics ?? []).forEach((pt: Record<string, unknown>) => {
   263→    const postId = pt.post_id as string;
   264→    const t = pt.topics as Topic;
   265→    if (!topicMap.has(postId)) topicMap.set(postId, []);
   266→    topicMap.get(postId)!.push(t);
   267→  });
   268→
   269→  // Fetch current user's votes
   270→  let voteMap = new Map<string, 1 | -1 | 0>();
   271→  if (currentUserId) {
   272→    const { data: votes } = await supabase!
   273→      .from("post_votes")
   274→      .select("post_id, value")
   275→      .eq("user_id", currentUserId)
   276→      .in("post_id", postIds);
   277→    (votes ?? []).forEach((v: { post_id: string; value: 1 | -1 }) => {
   278→      voteMap.set(v.post_id, v.value);
   279→    });
   280→  }
   281→
   282→  // Fetch current user's bookmarks
   283→  let bookmarkSet = new Set<string>();
   284→  if (currentUserId) {
   285→    const { data: bookmarks } = await supabase!
   286→      .from("bookmarks")
   287→      .select("post_id")
   288→      .eq("user_id", currentUserId)
   289→      .in("post_id", postIds);
   290→    (bookmarks ?? []).forEach((b: { post_id: string }) => bookmarkSet.add(b.post_id));
   291→  }
   292→
   293→  const posts: Post[] = rows.map((r) => {
   294→    const upvoteRow = (r.upvote_count as { count: number }[] | null)?.[0];
   295→    const downvoteRow = (r.downvote_count as { count: number }[] | null)?.[0];
   296→    const commentRow = (r.comment_count as { count: number }[] | null)?.[0];
   297→    return {
   298→      id: r.id,
   299→      author_id: r.author_id,
   300→      title: r.title,
   301→      preview: r.preview,
   302→      content: r.content,
   303→      images: (r.images as { url: string; alt?: string }[]) ?? [],
   304→      tags: (r.tags as string[]) ?? [],
   305→      views: r.views as number,
   306→      removed: r.removed as boolean,
   307→      removed_reason: r.removed_reason as string | null,
   308→      created_at: r.created_at,
   309→      updated_at: r.updated_at as string | null,
   310→      topic_ids: ((r.post_topics as { topic_id: string }[]) ?? []).map((pt) => pt.topic_id),
   311→      topics: topicMap.get(r.id as string) ?? [],
   312→      author: authorMap.get(r.author_id as string),
   313→      upvote_count: upvoteRow?.count ?? 0,
   314→      downvote_count: downvoteRow?.count ?? 0,
   315→      comment_count: commentRow?.count ?? 0,
   316→      my_vote: voteMap.get(r.id as string) ?? 0,
   317→      is_bookmarked: bookmarkSet.has(r.id as string),
   318→    };
   319→  });
   320→
   321→  // Sort trending/popular/week
   322→  const day = 24 * 60 * 60 * 1000;
   323→  if (sort === "popular") {
   324→    posts.sort((a, b) => (b.upvote_count - b.downvote_count) - (a.upvote_count - a.downvote_count));
   325→  } else if (sort === "trending" || sort === "week") {
   326→    posts.sort((a, b) => {
   327→      const ageA = (Date.now() - new Date(a.created_at).getTime()) / day;
   328→      const ageB = (Date.now() - new Date(b.created_at).getTime()) / day;
   329→      const scoreA = (a.upvote_count - a.downvote_count + a.comment_count * 2 + a.views / 100) / Math.pow(ageA + 2, 1.3);
   330→      const scoreB = (b.upvote_count - b.downvote_count + b.comment_count * 2 + b.views / 100) / Math.pow(ageB + 2, 1.3);
   331→      return scoreB - scoreA;
   332→    });
   333→  }
   334→
   335→  return posts.slice(0, limit);
   336→}

   338→export async function fetchPost(postId: string, currentUserId?: string): Promise<Post | null> {
   339→  if (!supabase) return null;
   340→  const { data, error } = await supabase
   341→    .from("posts")
   342→    .select("*")
   343→    .eq("id", postId)
   344→    .maybeSingle();
   345→  if (error || !data) return null;
   346→  // Increment view count (best-effort)
   347→  await supabase.rpc("increment_post_views", { post_id: postId }).catch(() => {});
   348→  // Use fetchPosts to get the fully transformed post
   349→  const posts = await fetchPosts({ sort: "latest", limit: 1, currentUserId });
   350→  // Actually we need to fetch by ID — let me do it manually
   351→  return transformPosts([data], "latest", 1, currentUserId).then((arr) => arr[0] ?? null);
   352→}

   354→export async function createPost(
   355→  data: {
   356→    title: string;
   357→    content: string;
   358→    topicIds: string[];
   359→    tags?: string[];
   360→    images?: { url: string; alt?: string }[];
   361→  },
   362→  authorId: string
   363→): Promise<string | null> {
   364→  if (!supabase) return null;
   365→  const preview = data.content.replace(/[#>*`_~\-\[\]\(\)!]/g, "").slice(0, 160).trim();
   366→  const { data: post, error } = await supabase
   367→    .from("posts")
   368→    .insert({
   369→      author_id: authorId,
   370→      title: data.title,
   371→      preview,
   372→      content: data.content,
   373→      tags: data.tags ?? [],
   374→      images: data.images ?? [],
   375→    })
   376→    .select("id")
   377→    .single();
   378→  if (error || !post) {
   379→    console.error("createPost error:", error);
   380→    return null;
   381→  }
   382→  // Link topics
   383→  if (data.topicIds.length > 0) {
   384→    const links = data.topicIds.map((topic_id) => ({ post_id: post.id, topic_id }));
   385→    await supabase.from("post_topics").insert(links);
   386→  }
   387→  return post.id;
   388→}

   390→export async function updatePost(
   391→  postId: string,
   392→  data: { title?: string; content?: string; topicIds?: string[]; tags?: string[] }
   393→): Promise<void> {
   394→  if (!supabase) return;
   395→  const update: Record<string, unknown> = { updated_at: new Date().toISOString() };
   396→  if (data.title !== undefined) update.title = data.title;
   397→  if (data.content !== undefined) {
   398→    update.content = data.content;
   399→    update.preview = data.content.replace(/[#>*`_~\-\[\]\(\)!]/g, "").slice(0, 160).trim();
   400→  }
   401→  if (data.tags !== undefined) update.tags = data.tags;
   402→  await supabase.from("posts").update(update).eq("id", postId);
   403→
   404→  if (data.topicIds !== undefined) {
   405→    await supabase.from("post_topics").delete().eq("post_id", postId);
   406→    if (data.topicIds.length > 0) {
   407→      const links = data.topicIds.map((topic_id) => ({ post_id: postId, topic_id }));
   408→      await supabase.from("post_topics").insert(links);
   409→    }
   410→  }
   411→}

   418→export async function fetchRandomPost(currentUserId?: string): Promise<Post | null> {
   419→  if (!supabase) return null;
   420→  const { count } = await supabase
   421→    .from("posts")
   422→    .select("id", { count: "exact", head: true })
   423→    .eq("removed", false);
   424→  if (!count || count === 0) return null;
   425→  const offset = Math.floor(Math.random() * count);
   426→  const { data } = await supabase
   427→    .from("posts")
   428→    .select("id, author_id, title, preview, content, images, tags, views, removed, removed_reason, created_at, updated_at")
   429→    .eq("removed", false)

   431→  if (!data || data.length === 0) return null;

   433→  return posts[0] ?? null;

   462→export async function fetchComments(postId: string, currentUserId?: string): Promise<Comment[]> {
   463→  if (!supabase) return [];
   464→  const { data, error } = await supabase
   465→    .from("comments")
   466→    .select("*")
   467→    .eq("post_id", postId)
   468→    .eq("removed", false)
   469→    .order("created_at", { ascending: true });
   470→  if (error || !data) return [];
   471→
   472→  const authorIds = [...new Set(data.map((c: { author_id: string }) => c.author_id))];
   473→  const commentIds = data.map((c: { id: string }) => c.id);
   474→
   475→  const { data: authors } = await supabase
   476→    .from("users")
   477→    .select("*")
   478→    .in("id", authorIds);
   479→  const authorMap = new Map<string, Profile>(
   480→    (authors ?? []).map((a: Profile) => [a.id, a])
   481→  );
   482→
   483→  let voteMap = new Map<string, 1 | -1 | 0>();
   484→  if (currentUserId) {
   485→    const { data: votes } = await supabase
   486→      .from("comment_votes")
   487→      .select("comment_id, value")
   488→      .eq("user_id", currentUserId)
   489→      .in("comment_id", commentIds);
   490→    (votes ?? []).forEach((v: { comment_id: string; value: 1 | -1 }) => {
   491→      voteMap.set(v.comment_id, v.value);
   492→    });
   493→  }
   494→
   495→  const { data: upvoteCounts } = await supabase
   496→    .from("comment_votes")
   497→    .select("comment_id, value")
   498→    .in("comment_id", commentIds);
   499→
   500→  const upMap = new Map<string, number>();
   501→  const downMap = new Map<string, number>();
   502→  (upvoteCounts ?? []).forEach((v: { comment_id: string; value: number }) => {
   503→    if (v.value === 1) upMap.set(v.comment_id, (upMap.get(v.comment_id) ?? 0) + 1);
   504→    else downMap.set(v.comment_id, (downMap.get(v.comment_id) ?? 0) + 1);
   505→  });
   506→
   507→  const comments: Comment[] = data.map((c: Record<string, unknown>) => ({
   508→    id: c.id as string,
   509→    post_id: c.post_id as string,
   510→    author_id: c.author_id as string,
   511→    parent_id: c.parent_id as string | null,
   512→    content: c.content as string,
   513→    mentions: (c.mentions as string[]) ?? [],
   514→    removed: c.removed as boolean,
   515→    created_at: c.created_at as string,
   516→    updated_at: c.updated_at as string | null,
   517→    author: authorMap.get(c.author_id as string),
   518→    upvote_count: upMap.get(c.id as string) ?? 0,
   519→    downvote_count: downMap.get(c.id as string) ?? 0,
   520→    my_vote: voteMap.get(c.id as string) ?? 0,
   521→  }));
   522→
   523→  // Build nested tree
   524→  const byId = new Map(comments.map((c) => [c.id, c]));
   525→  const roots: Comment[] = [];
   526→  comments.forEach((c) => {
   527→    if (c.parent_id && byId.has(c.parent_id)) {
   528→      const parent = byId.get(c.parent_id)!;
   529→      if (!parent.children) parent.children = [];
   530→      parent.children.push(c);
   531→    } else {
   532→      roots.push(c);
   533→    }
   534→  });
   535→  // Sort roots by score, children by time
   536→  roots.sort((a, b) => (b.upvote_count - b.downvote_count) - (a.upvote_count - a.downvote_count));
   537→  comments.forEach((c) => {
   538→    if (c.children) c.children.sort((a, b) => new Date(a.created_at).getTime() - new Date(b.created_at).getTime());
   539→  });
   540→  return roots;
   541→}

   543→export async function addComment(
   544→  data: { postId: string; parentId: string | null; content: string; mentions?: string[] },
   545→  authorId: string
   546→): Promise<string | null> {
   547→  if (!supabase) return null;
   548→  const { data: comment, error } = await supabase

   741→  posts: Post[];
   742→  topics: Topic[];

   745→
   746→export async function searchAll(
   747→  query: string,
   748→  filter: "all" | "posts" | "topics" | "users" = "all",
   749→  currentUserId?: string
   750→): Promise<SearchResults> {
   751→  if (!supabase || !query.trim()) return { posts: [], topics: [], users: [] };
   752→  const q = query.trim();
   753→
   754→  const [postsP, topicsP, usersP] = await Promise.all([
   755→    filter === "all" || filter === "posts"
   756→      ? supabase
   757→          .from("posts")
   758→          .select("id, author_id, title, preview, content, images, tags, views, removed, removed_reason, created_at, updated_at")
   759→          .or(`title.ilike.%${q}%,preview.ilike.%${q}%`)
   760→          .eq("removed", false)
   761→          .order("created_at", { ascending: false })
   762→          .limit(20)
   763→      : Promise.resolve({ data: [], error: null }),
   764→    filter === "all" || filter === "topics"
   765→      ? supabase
   766→          .from("topics")
   767→          .select("*")
   768→          .or(`name.ilike.%${q}%,description.ilike.%${q}%`)
   769→          .limit(20)
   770→      : Promise.resolve({ data: [], error: null }),
   771→    filter === "all" || filter === "users"
   772→      ? supabase
   773→          .from("users")
   774→          .select("*")
   775→          .or(`name.ilike.%${q}%,username.ilike.%${q}%,bio.ilike.%${q}%`)
   776→          .eq("banned", false)
   777→          .limit(20)
   778→      : Promise.resolve({ data: [], error: null }),
   779→  ]);
   780→
   781→  const posts = postsP.data ? await transformPosts(postsP.data as Record<string, unknown>[], "latest", 20, currentUserId) : [];
   782→  return {
   783→    posts,
   784→    topics: (topicsP.data ?? []) as Topic[],
   785→    users: (usersP.data ?? []) as Profile[],
   786→  };
   787→}
