   153→   153→export type FeedSort = "trending" | "latest" | "popular";
   154→   154→
   155→   155→interface FetchPostsOptions {
   156→   156→  sort?: FeedSort;
   157→   157→  topicId?: string;
   158→   158→  authorId?: string;
   159→   159→  bookmarkedBy?: string;
   160→   160→  limit?: number;
   161→   161→  offset?: number;
   162→   162→  currentUserId?: string;
   163→   163→}
   164→   164→
   165→   165→export async function fetchPosts(opts: FetchPostsOptions = {}): Promise<Post[]> {
   166→   166→  if (!supabase) return [];
   167→   167→  const { sort = "latest", topicId, authorId, bookmarkedBy, limit = 20, offset = 0, currentUserId } = opts;
   168→   168→
   169→   169→  let query = supabase.from("posts").select(`
   170→   170→      id, author_id, title, preview, content, images, tags, views,
   171→   171→      removed, removed_reason, created_at, updated_at,
   172→   172→      upvote_count:post_votes!post_votes_post_id_fkey(count),
   173→   173→      downvote_count:post_votes!post_votes_post_id_fkey(count),
   174→   174→      comment_count:comments!comments_post_id_fkey(count),
   175→   175→      post_topics(topic_id)
   176→   176→    `);
   177→   177→
   178→   178→  // Joins via post_topics
   179→   179→  if (topicId) {
   180→   180→    // Use a sub-filter — Supabase JS supports filtering on related tables
   181→   181→    query = supabase
   182→   182→      .from("posts")
   183→   183→      .select(`
   184→   184→        id, author_id, title, preview, content, images, tags, views,
   185→   185→        removed, removed_reason, created_at, updated_at,
   186→   186→        post_topics!inner(topic_id)
   187→   187→      `)
   188→   188→      .eq("post_topics.topic_id", topicId);
   189→   189→  }
   190→   190→
   191→   191→  if (authorId) query = query.eq("author_id", authorId);
   192→   192→
   193→   193→  // Note: bookmarkedBy filter is handled separately below because we can't
   194→   194→  // directly filter by a foreign key on the bookmarks table from posts.
   195→   195→
   196→   196→  let postsQuery = query.eq("removed", false);
   197→   197→
   198→   198→  // For "week" filter, restrict to last 7 days
   199→   199→  if (sort === "week") {
   200→   200→    const weekAgo = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000).toISOString();
   201→   201→    postsQuery = postsQuery.gte("created_at", weekAgo);
   202→   202→  }
   203→   203→
   204→   204→  if (sort === "latest") {
   205→   205→    postsQuery = postsQuery.order("created_at", { ascending: false });
   206→   206→  } else if (sort === "popular") {
   207→   207→    // Popular = most upvotes — we'll sort client-side after fetching counts
   208→   208→    postsQuery = postsQuery.order("created_at", { ascending: false }).limit(limit * 3);
   209→   209→  } else {
   210→   210→    // trending, week — fetch recent, sort client-side
   211→   211→    postsQuery = postsQuery.order("created_at", { ascending: false }).limit(limit * 3);
   212→   212→  }
   213→   213→
   214→   214→  postsQuery = postsQuery.range(offset, offset + limit - 1);
   215→   215→
   216→   216→  const { data, error } = await postsQuery;
   217→   217→  if (error) {
   218→   218→    console.error("fetchPosts error:", error);
   219→   219→    return [];
   220→   220→  }
   221→   221→
   222→   222→  // If filtering by bookmarked posts, fetch bookmarked post IDs first
   223→   223→  if (bookmarkedBy) {
   224→   224→    const { data: bookmarkData } = await supabase
   225→   225→      .from("bookmarks")
   226→   226→      .select("post_id")
   227→   227→      .eq("user_id", bookmarkBy);
   228→   228→    const bookmarkedIds = (bookmarkData ?? []).map((b: { post_id: string }) => b.post_id);
   229→   229→    const filtered = (data ?? []).filter((p: { id: string }) => bookmarkedIds.includes(p.id));
   230→   230→    return transformPosts(filtered, sort, limit, currentUserId);
   231→   231→  }
   232→   232→
   233→   233→  return transformPosts(data ?? [], sort, limit, currentUserId);
   234→   234→}
   235→   235→
   236→   236→async function transformPosts(
   237→   237→  rows: Record<string, unknown>[],
   238→   238→  sort: FeedSort,
   239→   239→  limit: number,
   240→   240→  currentUserId?: string
   241→   241→): Promise<Post[]> {
   242→   242→  if (rows.length === 0) return [];
   243→   243→
   244→   244→  const authorIds = [...new Set(rows.map((r) => r.author_id as string))];
   245→   245→  const postIds = rows.map((r) => r.id as string);
   246→   246→
   247→   247→  // Fetch authors
   248→   248→  const { data: authors } = await supabase!
   249→   249→    .from("users")
   250→   250→    .select("*")
   251→   251→    .in("id", authorIds);
   252→   252→  const authorMap = new Map<string, Profile>(
   253→   253→    (authors ?? []).map((a: Profile) => [a.id, a])
   254→   254→  );
   255→   255→
   256→   256→  // Fetch topics for all posts
   257→   257→  const { data: postTopics } = await supabase!
   258→   258→    .from("post_topics")
   259→   259→    .select("post_id, topic_id, topics!inner(id, name, slug, description, banner, color, icon, parent_id, post_count, created_at)")
   260→   260→    .in("post_id", postIds);
   261→   261→  const topicMap = new Map<string, Topic[]>();
   262→   262→  (postTopics ?? []).forEach((pt: Record<string, unknown>) => {
   263→   263→    const postId = pt.post_id as string;
   264→   264→    const t = pt.topics as Topic;
   265→   265→    if (!topicMap.has(postId)) topicMap.set(postId, []);
   266→   266→    topicMap.get(postId)!.push(t);
   267→   267→  });
   268→   268→
   269→   269→  // Fetch current user's votes
   270→   270→  let voteMap = new Map<string, 1 | -1 | 0>();
   271→   271→  if (currentUserId) {
   272→   272→    const { data: votes } = await supabase!
   273→   273→      .from("post_votes")
   274→   274→      .select("post_id, value")
   275→   275→      .eq("user_id", currentUserId)
   276→   276→      .in("post_id", postIds);
   277→   277→    (votes ?? []).forEach((v: { post_id: string; value: 1 | -1 }) => {
   278→   278→      voteMap.set(v.post_id, v.value);
   279→   279→    });
   280→   280→  }
   281→   281→
   282→   282→  // Fetch current user's bookmarks
   283→   283→  let bookmarkSet = new Set<string>();
   284→   284→  if (currentUserId) {
   285→   285→    const { data: bookmarks } = await supabase!
   286→   286→      .from("bookmarks")
   287→   287→      .select("post_id")
   288→   288→      .eq("user_id", currentUserId)
   289→   289→      .in("post_id", postIds);
   290→   290→    (bookmarks ?? []).forEach((b: { post_id: string }) => bookmarkSet.add(b.post_id));
   291→   291→  }
   292→   292→
   293→   293→  const posts: Post[] = rows.map((r) => {
   294→   294→    const upvoteRow = (r.upvote_count as { count: number }[] | null)?.[0];
   295→   295→    const downvoteRow = (r.downvote_count as { count: number }[] | null)?.[0];
   296→   296→    const commentRow = (r.comment_count as { count: number }[] | null)?.[0];
   297→   297→    return {
   298→   298→      id: r.id,
   299→   299→      author_id: r.author_id,
   300→   300→      title: r.title,
   301→   301→      preview: r.preview,
   302→   302→      content: r.content,
   303→   303→      images: (r.images as { url: string; alt?: string }[]) ?? [],
   304→   304→      tags: (r.tags as string[]) ?? [],
   305→   305→      views: r.views as number,
   306→   306→      removed: r.removed as boolean,
   307→   307→      removed_reason: r.removed_reason as string | null,
   308→   308→      created_at: r.created_at,
   309→   309→      updated_at: r.updated_at as string | null,
   310→   310→      topic_ids: ((r.post_topics as { topic_id: string }[]) ?? []).map((pt) => pt.topic_id),
   311→   311→      topics: topicMap.get(r.id as string) ?? [],
   312→   312→      author: authorMap.get(r.author_id as string),
   313→   313→      upvote_count: upvoteRow?.count ?? 0,
   314→   314→      downvote_count: downvoteRow?.count ?? 0,
   315→   315→      comment_count: commentRow?.count ?? 0,
   316→   316→      my_vote: voteMap.get(r.id as string) ?? 0,
   317→   317→      is_bookmarked: bookmarkSet.has(r.id as string),
   318→   318→    };
   319→   319→  });
   320→   320→
   321→   321→  // Sort trending/popular/week
   322→   322→  const day = 24 * 60 * 60 * 1000;
   323→   323→  if (sort === "popular") {
   324→   324→    posts.sort((a, b) => (b.upvote_count - b.downvote_count) - (a.upvote_count - a.downvote_count));
   325→   325→  } else if (sort === "trending" || sort === "week") {
   326→   326→    posts.sort((a, b) => {
   327→   327→      const ageA = (Date.now() - new Date(a.created_at).getTime()) / day;
   328→   328→      const ageB = (Date.now() - new Date(b.created_at).getTime()) / day;
   329→   329→      const scoreA = (a.upvote_count - a.downvote_count + a.comment_count * 2 + a.views / 100) / Math.pow(ageA + 2, 1.3);
   330→   330→      const scoreB = (b.upvote_count - b.downvote_count + b.comment_count * 2 + b.views / 100) / Math.pow(ageB + 2, 1.3);
   331→   331→      return scoreB - scoreA;
   332→   332→    });
   333→   333→  }
   334→   334→
   335→   335→  return posts.slice(0, limit);
   336→   336→}

   338→   338→export async function fetchPost(postId: string, currentUserId?: string): Promise<Post | null> {
   339→   339→  if (!supabase) return null;
   340→   340→  const { data, error } = await supabase
   341→   341→    .from("posts")
   342→   342→    .select("*")
   343→   343→    .eq("id", postId)
   344→   344→    .maybeSingle();
   345→   345→  if (error || !data) return null;
   346→   346→  // Increment view count (best-effort)
   347→   347→  await supabase.rpc("increment_post_views", { post_id: postId }).catch(() => {});
   348→   348→  // Use fetchPosts to get the fully transformed post
   349→   349→  const posts = await fetchPosts({ sort: "latest", limit: 1, currentUserId });
   350→   350→  // Actually we need to fetch by ID — let me do it manually
   351→   351→  return transformPosts([data], "latest", 1, currentUserId).then((arr) => arr[0] ?? null);
   352→   352→}

   354→   354→export async function createPost(
   355→   355→  data: {
   356→   356→    title: string;
   357→   357→    content: string;
   358→   358→    topicIds: string[];
   359→   359→    tags?: string[];
   360→   360→    images?: { url: string; alt?: string }[];
   361→   361→  },
   362→   362→  authorId: string
   363→   363→): Promise<string | null> {
   364→   364→  if (!supabase) return null;
   365→   365→  const preview = data.content.replace(/[#>*`_~\-\[\]\(\)!]/g, "").slice(0, 160).trim();
   366→   366→  const { data: post, error } = await supabase
   367→   367→    .from("posts")
   368→   368→    .insert({
   369→   369→      author_id: authorId,
   370→   370→      title: data.title,
   371→   371→      preview,
   372→   372→      content: data.content,
   373→   373→      tags: data.tags ?? [],
   374→   374→      images: data.images ?? [],
   375→   375→    })
   376→   376→    .select("id")
   377→   377→    .single();
   378→   378→  if (error || !post) {
   379→   379→    console.error("createPost error:", error);
   380→   380→    return null;
   381→   381→  }
   382→   382→  // Link topics
   383→   383→  if (data.topicIds.length > 0) {
   384→   384→    const links = data.topicIds.map((topic_id) => ({ post_id: post.id, topic_id }));
   385→   385→    await supabase.from("post_topics").insert(links);
   386→   386→  }
   387→   387→  return post.id;
   388→   388→}

   392→   392→  data: { title?: string; content?: string; topicIds?: string[]; tags?: string[] }
   393→   393→): Promise<void> {
   394→   394→  if (!supabase) return;

   396→   396→  if (data.title !== undefined) update.title = data.title;

   399→   399→    update.preview = data.content.replace(/[#>*`_~\-\[\]\(\)!]/g, "").slice(0, 160).trim();
   400→   400→  }

   403→   403→
   404→   404→  if (data.topicIds !== undefined) {
   405→   405→    await supabase.from("post_topics").delete().eq("post_id", postId);

   418→   418→export async function fetchRandomPost(currentUserId?: string): Promise<Post | null> {

   421→   421→    .from("posts")

   423→   423→    .eq("removed", false);
   424→   424→  if (!count || count === 0) return null;
   425→   425→  const offset = Math.floor(Math.random() * count);
   426→   426→  const { data } = await supabase
   427→   427→    .from("posts")
   428→   428→    .select("id, author_id, title, preview, content, images, tags, views, removed, removed_reason, created_at, updated_at")

   431→   431→  if (!data || data.length === 0) return null;

   463→   463→  if (!supabase) return [];
   464→   464→  const { data, error } = await supabase
   465→   465→    .from("comments")
   466→   466→    .select("*")
   467→   467→    .eq("post_id", postId)
   468→   468→    .eq("removed", false)
   469→   469→    .order("created_at", { ascending: true });
   470→   470→  if (error || !data) return [];
   471→   471→
   472→   472→  const authorIds = [...new Set(data.map((c: { author_id: string }) => c.author_id))];
   473→   473→  const commentIds = data.map((c: { id: string }) => c.id);
   474→   474→
   475→   475→  const { data: authors } = await supabase
   476→   476→    .from("users")
   477→   477→    .select("*")
   478→   478→    .in("id", authorIds);
   479→   479→  const authorMap = new Map<string, Profile>(
   480→   480→    (authors ?? []).map((a: Profile) => [a.id, a])
   481→   481→  );
   482→   482→
   483→   483→  let voteMap = new Map<string, 1 | -1 | 0>();
   484→   484→  if (currentUserId) {
   485→   485→    const { data: votes } = await supabase
   486→   486→      .from("comment_votes")
   487→   487→      .select("comment_id, value")
   488→   488→      .eq("user_id", currentUserId)
   489→   489→      .in("comment_id", commentIds);
   490→   490→    (votes ?? []).forEach((v: { comment_id: string; value: 1 | -1 }) => {
   491→   491→      voteMap.set(v.comment_id, v.value);
   492→   492→    });
   493→   493→  }
   494→   494→
   495→   495→  const { data: upvoteCounts } = await supabase
   496→   496→    .from("comment_votes")
   497→   497→    .select("comment_id, value")
   498→   498→    .in("comment_id", commentIds);
   499→   499→
   500→   500→  const upMap = new Map<string, number>();
   501→   501→  const downMap = new Map<string, number>();
   502→   502→  (upvoteCounts ?? []).forEach((v: { comment_id: string; value: number }) => {
   503→   503→    if (v.value === 1) upMap.set(v.comment_id, (upMap.get(v.comment_id) ?? 0) + 1);
   504→   504→    else downMap.set(v.comment_id, (downMap.get(v.comment_id) ?? 0) + 1);
   505→   505→  });
   506→   506→
   507→   507→  const comments: Comment[] = data.map((c: Record<string, unknown>) => ({
   508→   508→    id: c.id as string,
   509→   509→    post_id: c.post_id as string,
   510→   510→    author_id: c.author_id as string,
   511→   511→    parent_id: c.parent_id as string | null,
   512→   512→    content: c.content as string,
   513→   513→    mentions: (c.mentions as string[]) ?? [],
   514→   514→    removed: c.removed as boolean,
   515→   515→    created_at: c.created_at as string,
   516→   516→    updated_at: c.updated_at as string | null,
   517→   517→    author: authorMap.get(c.author_id as string),
   518→   518→    upvote_count: upMap.get(c.id as string) ?? 0,
   519→   519→    downvote_count: downMap.get(c.id as string) ?? 0,
   520→   520→    my_vote: voteMap.get(c.id as string) ?? 0,
   521→   521→  }));
   522→   522→
   523→   523→  // Build nested tree
   524→   524→  const byId = new Map(comments.map((c) => [c.id, c]));
   525→   525→  const roots: Comment[] = [];
   526→   526→  comments.forEach((c) => {
   527→   527→    if (c.parent_id && byId.has(c.parent_id)) {
   528→   528→      const parent = byId.get(c.parent_id)!;
   529→   529→      if (!parent.children) parent.children = [];
   530→   530→      parent.children.push(c);
   531→   531→    } else {
   532→   532→      roots.push(c);
   533→   533→    }
   534→   534→  });
   535→   535→  // Sort roots by score, children by time
   536→   536→  roots.sort((a, b) => (b.upvote_count - b.downvote_count) - (a.upvote_count - a.downvote_count));
   537→   537→  comments.forEach((c) => {
   538→   538→    if (c.children) c.children.sort((a, b) => new Date(a.created_at).getTime() - new Date(b.created_at).getTime());
   539→   539→  });
   540→   540→  return roots;
   541→   541→}
