   371→function FollowButton({ userId }: { userId: string }) {
   372→  const { profile } = useAuth();
   373→  const [following, setFollowing] = React.useState(false);
   374→
   375→  React.useEffect(() => {
   376→    if (!profile || !supabase) return;
   377→    let mounted = true;
   378→    (async () => {
   379→      const { data } = await supabase!
   380→        .from("user_followers")
   381→        .select("follower_id")
   382→        .eq("follower_id", profile.id)
   383→        .eq("followee_id", userId)
   384→        .maybeSingle();
   385→      if (mounted) setFollowing(!!data);
   386→    })();
   387→    return () => { mounted = false; };
   388→  }, [profile, userId]);
   389→
   390→  if (!profile || profile.id === userId) return null;
   391→
   392→  return (
   393→    <Button
   394→      variant={following ? "secondary" : "outline"}
   395→      size="sm"
   396→      className="rounded-lg text-xs"
   397→      onClick={async () => {
   398→        const { followUser, unfollowUser } = await import("@/lib/data");
   399→        if (following) {
   400→          await unfollowUser(profile.id, userId);
   401→          setFollowing(false);
   402→          toast.success("Unfollowed");
   403→        } else {
   404→          await followUser(profile.id, userId);
   405→          setFollowing(true);
   406→          toast.success("Following");
   407→        }
   408→      }}
   409→    >
   410→      {following ? "Following" : "Follow"}
   411→    </Button>
   412→  );
   413→}

   415→function CommentItem({ comment, postId, depth, onReload }: { comment: Comment; postId: string; depth: number; onReload: () => Promise<void> }) {
   416→  const { profile } = useAuth();
   417→  const [localComment, setLocalComment] = React.useState(comment);
   418→  const [replyOpen, setReplyOpen] = React.useState(false);
   419→  const [editOpen, setEditOpen] = React.useState(false);
   420→  const [replyText, setReplyText] = React.useState("");
   421→  const [editText, setEditText] = React.useState(comment.content);
   422→
   423→  React.useEffect(() => setLocalComment(comment), [comment]);
   424→
   425→  const author = localComment.author;
   426→  const hasUpvoted = localComment.my_vote === 1;
   427→  const hasDownvoted = localComment.my_vote === -1;
   428→  const score = localComment.upvote_count - localComment.downvote_count;
   429→  const isAuthor = profile?.id === localComment.author_id;
   430→
   431→  const handleVote = async (value: 1 | -1) => {
   432→    if (!profile) { toast.error("Sign in to vote"); return; }
   433→    const prev = localComment;
   434→    const newValue = prev.my_vote === value ? 0 : value;
   435→    setLocalComment({
   436→      ...prev,
   437→      my_vote: newValue as 1 | -1 | 0,
   438→      upvote_count: prev.upvote_count + (newValue === 1 ? 1 : prev.my_vote === 1 ? -1 : 0),
   439→      downvote_count: prev.downvote_count + (newValue === -1 ? 1 : prev.my_vote === -1 ? -1 : 0),
   440→    });
   441→    try {
   442→      if (newValue === 0) {
   443→        const { supabase } = await import("@/lib/data");
   444→        await supabase!.from("comment_votes").delete().match({ comment_id: comment.id, user_id: profile.id });
   445→      } else {
   446→        await voteOnComment(comment.id, profile.id, value);
   447→      }
   448→    } catch (e) {
   449→      console.error(e);
   450→      setLocalComment(prev);
   451→    }
   452→  };
   453→
   454→  const submitReply = async () => {
   455→    if (!replyText.trim() || !profile) return;
   456→    const id = await addComment({ postId, parentId: comment.id, content: replyText.trim() }, profile.id);
   457→    if (id) {
   458→      setReplyText("");
   459→      setReplyOpen(false);
   460→      toast.success("Reply added");
   461→      await onReload();
   462→    }
   463→  };
   464→
   465→  return (
   466→    <motion.div
   467→      initial={{ opacity: 0, y: 4 }}
   468→      animate={{ opacity: 1, y: 0 }}
   469→      exit={{ opacity: 0 }}
   470→      className={`glass-card rounded-2xl p-4 ${depth > 0 ? "ml-4 lg:ml-6" : ""}`}
   471→      style={{ marginLeft: depth > 0 ? `${Math.min(depth, 4) * 16}px` : 0 }}
   472→    >
   473→      <div className="flex items-start gap-3">
   474→        <Avatar className="w-8 h-8">
   475→          {author?.avatar_url ? <AvatarImage src={author.avatar_url} alt={author?.name} /> : null}
   476→          <AvatarFallback>{author?.name?.[0]?.toUpperCase() ?? "?"}</AvatarFallback>
   477→        </Avatar>
   478→        <div className="flex-1 min-w-0">
   479→          <div className="flex items-center gap-2 mb-1">
   480→            <button
   481→              onClick={() => author && useUIStore.getState().setView({ name: "profile", userId: author.id, tab: "posts" })}
   482→              className="font-medium text-sm hover:text-primary transition-colors"
   483→            >
   484→              {author?.name ?? "Unknown"}
   485→            </button>
   486→            <span className="text-xs text-muted-foreground">@{author?.username ?? "unknown"}</span>
   487→            <span className="text-xs text-muted-foreground">·</span>
   488→            <span className="text-xs text-muted-foreground">{timeAgo(localComment.created_at)}</span>
   489→            {localComment.updated_at && <span className="text-xs text-muted-foreground italic">· edited</span>}
   490→          </div>
   491→
   492→          {editOpen ? (
   493→            <div className="space-y-2">
   494→              <Textarea
   495→                value={editText}
   496→                onChange={(e) => setEditText(e.target.value)}
   497→                rows={3}
   498→                className="rounded-lg resize-none bg-muted/40"
   499→              />
   500→              <div className="flex gap-2">
   501→                <Button size="sm" onClick={async () => { await updateComment(comment.id, editText); setEditOpen(false); toast.success("Edited"); }}>Save</Button>
   502→                <Button size="sm" variant="ghost" onClick={() => { setEditOpen(false); setEditText(localComment.content); }}>Cancel</Button>
   503→              </div>
   504→            </div>
   505→          ) : (
   506→            <div className="text-sm leading-relaxed prose-nexus">
   507→              <ReactMarkdown remarkPlugins={[remarkGfm]}>{localComment.content}</ReactMarkdown>
   508→            </div>
   509→          )}
   510→
   511→          <div className="flex items-center gap-1 mt-2">
   512→            <div className="inline-flex items-center gap-0.5 p-0.5 rounded-lg bg-muted/40">
   513→              <button onClick={() => handleVote(1)} className={`p-1 rounded hover:bg-accent transition-colors ${hasUpvoted ? "text-primary" : "text-muted-foreground hover:text-foreground"}`}>
   514→                <ArrowBigUp className={`w-3.5 h-3.5 ${hasUpvoted ? "fill-current" : ""}`} />
   515→              </button>
   516→              <span className={`text-xs font-semibold min-w-[20px] text-center ${hasUpvoted ? "text-primary" : hasDownvoted ? "text-destructive" : "text-foreground"}`}>{score}</span>
   517→              <button onClick={() => handleVote(-1)} className={`p-1 rounded hover:bg-accent transition-colors ${hasDownvoted ? "text-destructive" : "text-muted-foreground hover:text-foreground"}`}>
   518→                <ArrowBigDown className={`w-3.5 h-3.5 ${hasDownvoted ? "fill-current" : ""}`} />
   519→              </button>
   520→            </div>
   521→            <Button variant="ghost" size="sm" onClick={() => setReplyOpen((v) => !v)} className="h-7 px-2 text-xs gap-1">
   522→              <Reply className="w-3 h-3" /> Reply
   523→            </Button>
   524→            <DropdownMenu>
   525→              <DropdownMenuTrigger asChild>
   526→                <Button variant="ghost" size="icon" className="h-7 w-7">
   527→                  <MoreHorizontal className="w-3 h-3" />
   528→                </Button>
   529→              </DropdownMenuTrigger>
   530→              <DropdownMenuContent align="end">
   531→                {isAuthor && (
   532→                  <DropdownMenuItem onClick={() => setEditOpen(true)}>
   533→                    <Edit className="w-3.5 h-3.5 mr-2" /> Edit
   534→                  </DropdownMenuItem>
   535→                )}
   536→                <DropdownMenuItem onSelect={async () => {
   537→                  const reason = window.prompt("Report reason");
   538→                  if (reason && profile && supabase) {
   539→                    await supabase.from("reports").insert({ reporter_id: profile.id, target_type: "comment", target_id: comment.id, reason });
   540→                    toast.success("Reported");
   541→                  }
   542→                }}>
   543→                  <Flag className="w-3.5 h-3.5 mr-2" /> Report
   544→                </DropdownMenuItem>
   545→                {isAuthor && (
   546→                  <>
   547→                    <DropdownMenuSeparator />
   548→                    <AlertDialog>
   549→                      <AlertDialogTrigger asChild>
   550→                        <DropdownMenuItem onSelect={(e) => e.preventDefault()} className="text-destructive focus:text-destructive">
   551→                          <Trash2 className="w-3.5 h-3.5 mr-2" /> Delete
   552→                        </DropdownMenuItem>
   553→                      </AlertDialogTrigger>
   554→                      <AlertDialogContent>
   555→                        <AlertDialogHeader>
   556→                          <AlertDialogTitle>Delete this comment?</AlertDialogTitle>
   557→                          <AlertDialogDescription>This will remove the comment and all replies.</AlertDialogDescription>
   558→                        </AlertDialogHeader>
   559→                        <AlertDialogFooter>
   560→                          <AlertDialogCancel>Cancel</AlertDialogCancel>
   561→                          <AlertDialogAction
   562→                            onClick={async () => {
   563→                              await deleteComment(comment.id);
   564→                              toast.success("Deleted");
   565→                              await onReload();
   566→                            }}
   567→                            className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   568→                          >Delete</AlertDialogAction>
   569→                        </AlertDialogFooter>
   570→                      </AlertDialogContent>
   571→                    </AlertDialog>
   572→                  </>
   573→                )}
   574→              </DropdownMenuContent>
   575→            </DropdownMenu>
   576→          </div>
   577→
   578→          {replyOpen && (
   579→            <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: "auto" }} className="mt-3">
   580→              <Textarea
   581→                value={replyText}
   582→                onChange={(e) => setReplyText(e.target.value)}
   583→                placeholder={`Reply to @${author?.username ?? "user"}…`}
   584→                rows={2}
   585→                className="rounded-lg resize-none bg-muted/40 text-sm"
   586→                autoFocus
   587→              />
   588→              <div className="flex gap-2 mt-2">
   589→                <Button size="sm" onClick={submitReply} disabled={!replyText.trim()}>Reply</Button>
   590→                <Button size="sm" variant="ghost" onClick={() => setReplyOpen(false)}>Cancel</Button>
   591→              </div>
   592→            </motion.div>
   593→          )}
   594→
   595→          {localComment.children && localComment.children.length > 0 && (
   596→            <div className="mt-3 space-y-2 border-l-2 border-border/40 pl-3">
   597→              {localComment.children.map((child) => (
   598→                <CommentItem key={child.id} comment={child} postId={postId} depth={depth + 1} onReload={onReload} />
   599→              ))}
   600→            </div>
   601→          )}
   602→        </div>
   603→      </div>
   604→    </motion.div>
   605→  );
   606→}
