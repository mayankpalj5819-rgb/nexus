   380→   380→function FollowButton({ userId }: { userId: string }) {
   381→   381→  const { profile } = useAuth();
   382→   382→  const [following, setFollowing] = React.useState(false);
   383→   383→
   384→   384→  React.useEffect(() => {
   385→   385→    if (!profile || !supabase) return;
   386→   386→    let mounted = true;
   387→   387→    (async () => {
   388→   388→      const { data } = await supabase!
   389→   389→        .from("user_followers")
   390→   390→        .select("follower_id")
   391→   391→        .eq("follower_id", profile.id)
   392→   392→        .eq("followee_id", userId)
   393→   393→        .maybeSingle();
   394→   394→      if (mounted) setFollowing(!!data);
   395→   395→    })();
   396→   396→    return () => { mounted = false; };
   397→   397→  }, [profile, userId]);
   398→   398→
   399→   399→  if (!profile || profile.id === userId) return null;
   400→   400→
   401→   401→  return (
   402→   402→    <Button
   403→   403→      variant={following ? "secondary" : "outline"}
   404→   404→      size="sm"
   405→   405→      className="rounded-lg text-xs"
   406→   406→      onClick={async () => {
   407→   407→        const { followUser, unfollowUser } = await import("@/lib/data");
   408→   408→        if (following) {
   409→   409→          await unfollowUser(profile.id, userId);
   410→   410→          setFollowing(false);
   411→   411→          toast.success("Unfollowed");
   412→   412→        } else {
   413→   413→          await followUser(profile.id, userId);
