   380→function FollowButton({ userId }: { userId: string }) {
   381→  const { profile } = useAuth();
   382→  const [following, setFollowing] = React.useState(false);
   383→
   384→  React.useEffect(() => {
   385→    if (!profile || !supabase) return;
   386→    let mounted = true;
   387→    (async () => {
   388→      const { data } = await supabase!
   389→        .from("user_followers")
   390→        .select("follower_id")
   391→        .eq("follower_id", profile.id)
   392→        .eq("followee_id", userId)
   393→        .maybeSingle();
   394→      if (mounted) setFollowing(!!data);
   395→    })();
   396→    return () => { mounted = false; };
   397→  }, [profile, userId]);
   398→
   399→  if (!profile || profile.id === userId) return null;
   400→
   401→  return (
   402→    <Button
   403→      variant={following ? "secondary" : "outline"}
   404→      size="sm"
   405→      className="rounded-lg text-xs"
   406→      onClick={async () => {
   407→        const { followUser, unfollowUser } = await import("@/lib/data");
   408→        if (following) {
   409→          await unfollowUser(profile.id, userId);
   410→          setFollowing(false);
   411→          toast.success("Unfollowed");
   412→        } else {
   413→          await followUser(profile.id, userId);
   414→          setFollowing(true);
   415→          toast.success("Following");
   416→        }
   417→      }}
   418→    >
   419→      {following ? "Following" : "Follow"}
   420→    </Button>
   421→  );
   422→}

   424→function CommentItem({ comment, postId, depth, onReload }: { comment: Comment; postId: string; depth: number; onReload: () => Promise<void> }) {
   425→  const { profile } = useAuth();
   426→  const [localComment, setLocalComment] = React.useState(comment);
   427→  const [replyOpen, setReplyOpen] = React.useState(false);
   428→  const [editOpen, setEditOpen] = React.useState(false);
   429→  const [replyText, setReplyText] = React.useState("");
   430→  const [editText, setEditText] = React.useState(comment.content);
   431→
   432→  React.useEffect(() => setLocalComment(comment), [comment]);
   433→
   434→  const author = localComment.author;
   435→  const hasUpvoted = localComment.my_vote === 1;
   436→  const hasDownvoted = localComment.my_vote === -1;
   437→  const score = localComment.upvote_count - localComment.downvote_count;
   438→  const isAuthor = profile?.id === localComment.author_id;
   439→
   440→  const handleVote = async (value: 1 | -1) => {
   441→    if (!profile) { toast.error("Sign in to vote"); return; }
   442→    const prev = localComment;
   443→    const newValue = prev.my_vote === value ? 0 : value;
   444→    setLocalComment({
   445→      ...prev,
   446→      my_vote: newValue as 1 | -1 | 0,
   447→      upvote_count: prev.upvote_count + (newValue === 1 ? 1 : prev.my_vote === 1 ? -1 : 0),
   448→      downvote_count: prev.downvote_count + (newValue === -1 ? 1 : prev.my_vote === -1 ? -1 : 0),
   449→    });
   450→    try {
   451→      if (newValue === 0) {
   452→        const { supabase } = await import("@/lib/data");
   453→        await supabase!.from("comment_votes").delete().match({ comment_id: comment.id, user_id: profile.id });
   454→      } else {
   455→        await voteOnComment(comment.id, profile.id, value);
   456→      }
   457→    } catch (e) {
   458→      console.error(e);
   459→      setLocalComment(prev);
   460→    }
   461→  };
   462→
   463→  const submitReply = async () => {
   464→    if (!replyText.trim() || !profile) return;
   465→    const id = await addComment({ postId, parentId: comment.id, content: replyText.trim() }, profile.id);
   466→    if (id) {
   467→      setReplyText("");
   468→      setReplyOpen(false);
   469→      toast.success("Reply added");
   470→      await onReload();
   471→    }
   472→  };
   473→
   474→  return (
   475→    <motion.div
   476→      initial={{ opacity: 0, y: 4 }}
   477→      animate={{ opacity: 1, y: 0 }}
   478→      exit={{ opacity: 0 }}
   479→      className={`glass-card rounded-2xl p-4 ${depth > 0 ? "ml-4 lg:ml-6" : ""}`}
   480→      style={{ marginLeft: depth > 0 ? `${Math.min(depth, 4) * 16}px` : 0 }}
   481→    >
   482→      <div className="flex items-start gap-3">
   483→        <Avatar className="w-8 h-8">
   484→          {author?.avatar_url ? <AvatarImage src={author.avatar_url} alt={author?.name} /> : null}
   485→          <AvatarFallback>{author?.name?.[0]?.toUpperCase() ?? "?"}</AvatarFallback>
   486→        </Avatar>
   487→        <div className="flex-1 min-w-0">
   488→          <div className="flex items-center gap-2 mb-1">
   489→            <button
   490→              onClick={() => author && useUIStore.getState().setView({ name: "profile", userId: author.id, tab: "posts" })}
   491→              className="font-medium text-sm hover:text-primary transition-colors"
   492→            >
   493→              {author?.name ?? "Unknown"}
   494→            </button>
   495→            <span className="text-xs text-muted-foreground">@{author?.username ?? "unknown"}</span>
   496→            <span className="text-xs text-muted-foreground">·</span>
   497→            <span className="text-xs text-muted-foreground">{timeAgo(localComment.created_at)}</span>
   498→            {localComment.updated_at && <span className="text-xs text-muted-foreground italic">· edited</span>}
   499→          </div>
   500→
   501→          {editOpen ? (
   502→            <div className="space-y-2">
   503→              <Textarea
   504→                value={editText}
   505→                onChange={(e) => setEditText(e.target.value)}
   506→                rows={3}
   507→                className="rounded-lg resize-none bg-muted/40"
   508→              />
   509→              <div className="flex gap-2">
   510→                <Button size="sm" onClick={async () => { await updateComment(comment.id, editText); setEditOpen(false); toast.success("Edited"); }}>Save</Button>
   511→                <Button size="sm" variant="ghost" onClick={() => { setEditOpen(false); setEditText(localComment.content); }}>Cancel</Button>
   512→              </div>
   513→            </div>
   514→          ) : (
   515→            <div className="text-sm leading-relaxed prose-nexus">
   516→              <ReactMarkdown remarkPlugins={[remarkGfm]}>{localComment.content}</ReactMarkdown>
   517→            </div>
   518→          )}
   519→
   520→          <div className="flex items-center gap-1 mt-2">
   521→            <div className="inline-flex items-center gap-0.5 p-0.5 rounded-lg bg-muted/40">
   522→              <button onClick={() => handleVote(1)} className={`p-1 rounded hover:bg-accent transition-colors ${hasUpvoted ? "text-primary" : "text-muted-foreground hover:text-foreground"}`}>
   523→                <ArrowBigUp className={`w-3.5 h-3.5 ${hasUpvoted ? "fill-current" : ""}`} />
   524→              </button>
   525→              <span className={`text-xs font-semibold min-w-[20px] text-center ${hasUpvoted ? "text-primary" : hasDownvoted ? "text-destructive" : "text-foreground"}`}>{score}</span>
   526→              <button onClick={() => handleVote(-1)} className={`p-1 rounded hover:bg-accent transition-colors ${hasDownvoted ? "text-destructive" : "text-muted-foreground hover:text-foreground"}`}>
   527→                <ArrowBigDown className={`w-3.5 h-3.5 ${hasDownvoted ? "fill-current" : ""}`} />
   528→              </button>
   529→            </div>
   530→            <Button variant="ghost" size="sm" onClick={() => setReplyOpen((v) => !v)} className="h-7 px-2 text-xs gap-1">
   531→              <Reply className="w-3 h-3" /> Reply
   532→            </Button>
   533→            <DropdownMenu>
   534→              <DropdownMenuTrigger asChild>
   535→                <Button variant="ghost" size="icon" className="h-7 w-7">
   536→                  <MoreHorizontal className="w-3 h-3" />
   537→                </Button>
   538→              </DropdownMenuTrigger>
   539→              <DropdownMenuContent align="end">
   540→                {isAuthor && (
   541→                  <DropdownMenuItem onClick={() => setEditOpen(true)}>
   542→                    <Edit className="w-3.5 h-3.5 mr-2" /> Edit
   543→                  </DropdownMenuItem>
   544→                )}
   545→                <DropdownMenuItem onSelect={async () => {
   546→                  const reason = window.prompt("Report reason");
   547→                  if (reason && profile && supabase) {
   548→                    await supabase.from("reports").insert({ reporter_id: profile.id, target_type: "comment", target_id: comment.id, reason });
   549→                    toast.success("Reported");
   550→                  }
   551→                }}>
   552→                  <Flag className="w-3.5 h-3.5 mr-2" /> Report
   553→                </DropdownMenuItem>
   554→                {isAuthor && (
   555→                  <>
   556→                    <DropdownMenuSeparator />
   557→                    <AlertDialog>
   558→                      <AlertDialogTrigger asChild>
   559→                        <DropdownMenuItem onSelect={(e) => e.preventDefault()} className="text-destructive focus:text-destructive">
   560→                          <Trash2 className="w-3.5 h-3.5 mr-2" /> Delete
   561→                        </DropdownMenuItem>
   562→                      </AlertDialogTrigger>
   563→                      <AlertDialogContent>
   564→                        <AlertDialogHeader>
   565→                          <AlertDialogTitle>Delete this comment?</AlertDialogTitle>
   566→                          <AlertDialogDescription>This will remove the comment and all replies.</AlertDialogDescription>
   567→                        </AlertDialogHeader>
   568→                        <AlertDialogFooter>
   569→                          <AlertDialogCancel>Cancel</AlertDialogCancel>
   570→                          <AlertDialogAction
   571→                            onClick={async () => {
   572→                              await deleteComment(comment.id);
   573→                              toast.success("Deleted");
   574→                              await onReload();
   575→                            }}
   576→                            className="bg-destructive text-destructive-foreground hover:bg-destructive/90"
   577→                          >Delete</AlertDialogAction>
   578→                        </AlertDialogFooter>
   579→                      </AlertDialogContent>
   580→                    </AlertDialog>
   581→                  </>
   582→                )}
   583→              </DropdownMenuContent>
   584→            </DropdownMenu>
   585→          </div>
   586→
   587→          {replyOpen && (
   588→            <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: "auto" }} className="mt-3">
   589→              <Textarea
   590→                value={replyText}
   591→                onChange={(e) => setReplyText(e.target.value)}
   592→                placeholder={`Reply to @${author?.username ?? "user"}…`}
   593→                rows={2}
   594→                className="rounded-lg resize-none bg-muted/40 text-sm"
   595→                autoFocus
   596→              />
   597→              <div className="flex gap-2 mt-2">
   598→                <Button size="sm" onClick={submitReply} disabled={!replyText.trim()}>Reply</Button>
   599→                <Button size="sm" variant="ghost" onClick={() => setReplyOpen(false)}>Cancel</Button>
   600→              </div>
   601→            </motion.div>
   602→          )}
   603→
   604→          {localComment.children && localComment.children.length > 0 && (
   605→            <div className="mt-3 space-y-2 border-l-2 border-border/40 pl-3">
   606→              {localComment.children.map((child) => (
   607→                <CommentItem key={child.id} comment={child} postId={postId} depth={depth + 1} onReload={onReload} />
   608→              ))}
   609→            </div>
   610→          )}
   611→        </div>
   612→      </div>
   613→    </motion.div>
   614→  );
   615→}
